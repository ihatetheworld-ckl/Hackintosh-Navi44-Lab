cmake_minimum_required(VERSION 3.16)
option(NAVI44_MACOS26_TARGET "Use the current Tahoe CPU build profile (OFF only for legacy reference)" ON)
if(CMAKE_HOST_APPLE AND NAVI44_MACOS26_TARGET)
  if(NOT IS_DIRECTORY "${CMAKE_OSX_SYSROOT}")
    message(FATAL_ERROR "M13 requires an explicit existing SDK26.5 via CMAKE_OSX_SYSROOT; no auto selection/install")
  endif()
  file(READ "${CMAKE_OSX_SYSROOT}/SDKSettings.json" NAVI44_SDK_SETTINGS)
  if(NOT NAVI44_SDK_SETTINGS MATCHES "\"Version\"[ \t\r\n]*:[ \t\r\n]*\"26\\.5\"")
    message(FATAL_ERROR "M13 requires genuine SDK26.5; SDK27 is a separate negative/legacy reference")
  endif()
  set(CMAKE_OSX_DEPLOYMENT_TARGET "26.5" CACHE STRING "M13 static target" FORCE)
endif()
project(Navi44Offline LANGUAGES CXX)
# No install, fetching, IOKit, device.cpp, plugin.cpp, GPU API or hardware backend.
set(DIR "${CMAKE_CURRENT_SOURCE_DIR}/development/AMDRadeonRDNA4-Navi44/src")
set(CORE "${CMAKE_CURRENT_SOURCE_DIR}/development/AMDRadeonRDNA4-Navi44/core")
set(CORE_SOURCES "${DIR}/codeobj.cpp" "${DIR}/ipdiscovery.cpp" "${CORE}/validation.cpp" "${CORE}/device_info.cpp")
set(M8_SOURCES "${DIR}/edid.cpp" "${DIR}/atombios.cpp" "${CORE}/display.cpp" "${CORE}/integrity.cpp" "${CORE}/query_candidate.cpp")
set(LIFETIME_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/development/AMDRadeonRDNA4-Navi44/platform/lifecycle.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/development/AMDRadeonRDNA4-Navi44/platform/shutdown.cpp")
add_executable(navi44_host_tests tests/unit/m65_unit.cpp "${DIR}/codeobj.cpp" "${DIR}/ipdiscovery.cpp" "${DIR}/offlinevalidation.cpp" "${CORE}/validation.cpp" "${CORE}/device_info.cpp")
target_include_directories(navi44_host_tests PRIVATE "${DIR}")
target_compile_features(navi44_host_tests PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(navi44_host_tests PRIVATE /W4 /UNDEBUG)
else()
  target_compile_options(navi44_host_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
enable_testing()
add_test(NAME cpu_only COMMAND navi44_host_tests)
add_executable(navi44_probe tests/unit/m65_probe.cpp "${DIR}/offlinevalidation.cpp" ${CORE_SOURCES})
target_include_directories(navi44_probe PRIVATE "${DIR}")
target_compile_features(navi44_probe PRIVATE cxx_std_17)
add_executable(navi44_m7_tests tests/unit/m7_unit.cpp "${DIR}/offlinevalidation.cpp" ${CORE_SOURCES})
target_include_directories(navi44_m7_tests PRIVATE "${DIR}")
target_compile_features(navi44_m7_tests PRIVATE cxx_std_17)
add_test(NAME m7_cpu_only COMMAND navi44_m7_tests)
add_executable(navi44_m8_tests tests/unit/m8_unit.cpp ${CORE_SOURCES} ${M8_SOURCES})
add_executable(navi44_m8_fuzz tests/unit/m8_fuzz.cpp ${CORE_SOURCES} ${M8_SOURCES})
foreach(target navi44_m8_tests navi44_m8_fuzz)
  target_include_directories(${target} PRIVATE "${DIR}")
  target_compile_features(${target} PRIVATE cxx_std_17)
endforeach()
add_test(NAME m8_cpu_only COMMAND navi44_m8_tests)
add_test(NAME m8_bounded_mutation COMMAND navi44_m8_fuzz)
find_package(Threads REQUIRED)
add_executable(navi44_m10_lifetime tests/unit/m10_lifecycle.cpp ${CORE_SOURCES} ${LIFETIME_SOURCES})
target_compile_features(navi44_m10_lifetime PRIVATE cxx_std_17)
target_link_libraries(navi44_m10_lifetime PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(navi44_m10_lifetime PRIVATE /W4 /UNDEBUG)
else()
  target_compile_options(navi44_m10_lifetime PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME m10_lifetime_cpu_only COMMAND navi44_m10_lifetime)
add_executable(navi44_m11_lifetime tests/unit/m11_lifecycle.cpp ${LIFETIME_SOURCES})
target_compile_features(navi44_m11_lifetime PRIVATE cxx_std_17)
target_link_libraries(navi44_m11_lifetime PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(navi44_m11_lifetime PRIVATE /W4 /UNDEBUG)
else()
  target_compile_options(navi44_m11_lifetime PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME m11_lifetime_cpu_only COMMAND navi44_m11_lifetime)
add_executable(navi44_m12_lifetime tests/unit/m12_lifecycle.cpp ${LIFETIME_SOURCES})
target_compile_features(navi44_m12_lifetime PRIVATE cxx_std_17)
target_link_libraries(navi44_m12_lifetime PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(navi44_m12_lifetime PRIVATE /W4 /UNDEBUG)
else()
  target_compile_options(navi44_m12_lifetime PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME m12_lifetime_cpu_only COMMAND navi44_m12_lifetime)
set(M13_SOURCES "${CORE}/displayid.cpp" "${CORE}/driver_model.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/development/AMDRadeonRDNA4-Navi44/platform/resource_model.cpp")
add_executable(navi44_m13_models tests/unit/m13_models.cpp ${CORE_SOURCES} ${M8_SOURCES} ${LIFETIME_SOURCES} ${M13_SOURCES})
target_compile_features(navi44_m13_models PRIVATE cxx_std_17)
target_link_libraries(navi44_m13_models PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(navi44_m13_models PRIVATE /W4 /UNDEBUG)
  set_source_files_properties(${M13_SOURCES} PROPERTIES COMPILE_OPTIONS "/GR-;/EHs-c-")
else()
  target_compile_options(navi44_m13_models PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  set_source_files_properties(${M13_SOURCES} PROPERTIES COMPILE_OPTIONS "-fno-exceptions;-fno-rtti;-ffreestanding;-Wframe-larger-than=4096")
endif()
add_test(NAME m13_models_cpu_only COMMAND navi44_m13_models)
add_executable(navi44_m10_regression_proof tests/unit/m10_regression_proof.cpp)
target_include_directories(navi44_m10_regression_proof PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/development/AMDRadeonRDNA4-Navi44/platform")
target_compile_features(navi44_m10_regression_proof PRIVATE cxx_std_17)
add_test(NAME m10_regression_proof_cpu_only COMMAND navi44_m10_regression_proof)
set_tests_properties(cpu_only m7_cpu_only m8_cpu_only m8_bounded_mutation
  m10_lifetime_cpu_only m11_lifetime_cpu_only m12_lifetime_cpu_only m13_models_cpu_only m10_regression_proof_cpu_only PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
# Only Core translation units use these flags; Host probe retains its STL/exception API.
if(MSVC)
  set_source_files_properties(${CORE_SOURCES} ${M8_SOURCES} ${LIFETIME_SOURCES} PROPERTIES COMPILE_OPTIONS "/GR-;/EHs-c-")
else()
  set_source_files_properties(${CORE_SOURCES} ${M8_SOURCES} ${LIFETIME_SOURCES} PROPERTIES COMPILE_OPTIONS
    "-fno-exceptions;-fno-rtti;-ffreestanding;-Wframe-larger-than=4096")
  foreach(target navi44_probe navi44_m7_tests navi44_m8_tests navi44_m8_fuzz)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  endforeach()
endif()
